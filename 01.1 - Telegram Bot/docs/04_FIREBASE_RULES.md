## Rules Update

Thêm vào Firestore Security Rules hiện tại:

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ── Helper functions ──
    function isBotAccount() {
      return request.auth != null 
             && request.auth.token.role == "bot";
    }
    
    function isValidLicenseWrite() {
      // Bot chỉ được set các field cụ thể
      return request.resource.data.keys().hasOnly([
        '_t', '_st', '_cr', '_exp', '_mid', '_em', '_cn', 
        '_nt', '_dur', '_role', '_app', '_lim', '_lt_rolling',
        '_prev_key', '_client_submitted'
      ]);
    }
    
    // ════════════════════════════════════════
    // EXISTING RULES (giữ nguyên)
    // ════════════════════════════════════════
    
    // ... (rules hiện tại cho client, seller...)
    
    // ════════════════════════════════════════
    // BOT-SPECIFIC RULES
    // ════════════════════════════════════════
    
    // ── License keys ──
    match /_lic/{keyId} {
      allow read: if isBotAccount();
      allow create: if isBotAccount() && isValidLicenseWrite();
      allow update: if isBotAccount() 
                    && request.resource.data.diff(resource.data)
                       .affectedKeys()
                       .hasOnly(['_st', '_exp', '_dur', '_t', '_nt', 
                                 '_role', '_rv_at', '_rv_reason']);
      allow delete: if isBotAccount();
    }
    
    // ── Upgrade requests ──
    match /_upgrade_requests/{requestId} {
      allow read: if isBotAccount();
      allow update: if isBotAccount()
                    && request.resource.data.diff(resource.data)
                       .affectedKeys()
                       .hasOnly(['status', '_claimed_by', '_claimed_at']);
      allow delete: if isBotAccount();
    }
    
    // ── Trials ──
    match /_trials/{trialId} {
      allow read: if isBotAccount();
      allow write: if isBotAccount();
    }
    
    // ── MID to Key mapping ──
    match /_mid_to_key/{mid} {
      allow read: if isBotAccount();
      allow write: if isBotAccount();
    }
    
    // ── Customers ──
    match /_customers/{customerId} {
      allow read: if isBotAccount();
      allow write: if isBotAccount();
    }
    
    // ── Bot audit log (write-only) ──
    match /_bot_audit/{logId} {
      allow create: if isBotAccount();
      allow read: if false;  // Admin SDK only
    }
    
    // ── Config (read-only for bot) ──
    match /_config/{configId} {
      allow read: if isBotAccount();
    }
    
    // ── Rate limit (bot reads only) ──
    match /_rate_limit/{limitId} {
      allow read: if isBotAccount();
    }
  }
}
```

---

## Security Principles

| Principle | Implementation |
|-----------|---------------|
| **Least privilege** | Bot chỉ write fields cần thiết |
| **Field-level control** | `affectedKeys().hasOnly(...)` chặn thêm field lạ |
| **Audit trail** | `_bot_audit` collection ghi log mọi action |
| **No escalation** | Bot không thể modify Security Rules hoặc Auth settings |
| **Separation** | Bot có UID riêng, không dùng chung account admin |

```bash

curl -H "Authorization: Bearer <ID_TOKEN>" \
  "https://firestore.googleapis.com/v1/projects/<PROJECT>/databases/(default)/documents/_lic?pageSize=1"

# Test write to blocked collection (should 403)
curl -X PATCH \
  -H "Authorization: Bearer <ID_TOKEN>" \
  -H "Content-Type: application/json" \
  "https://firestore.googleapis.com/v1/projects/<PROJECT>/databases/(default)/documents/_sellers/test" \
  -d '{"fields":{"hack":{"stringValue":"test"}}}'
# Expected: 403 PERMISSION_DENIED ✅
```
